const CACHE = 'country-quiz-v1';

self.addEventListener('install', () => self.skipWaiting());

self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys()
      .then(keys => Promise.all(keys.filter(key => key !== CACHE).map(key => caches.delete(key))))
      .then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', event => {
  const { request } = event;
  if (request.method !== 'GET' || new URL(request.url).origin !== self.location.origin) return;
  event.respondWith(
    fetch(request)
      .then(response => {
        if (response.ok) {
          const copy = response.clone();
          event.waitUntil(caches.open(CACHE).then(cache => cache.put(request, copy)));
        }
        return response;
      })
      .catch(() => caches.match(request).then(cached => cached ?? caches.match('/')).then(cached => cached ?? Response.error()))
  );
});

import type { CountryRecord, Dataset, Metric, MetricId, Observation } from '../../src/types.ts';
import type { ObservationsByIso3 } from './observations.ts';

export interface SourceCountry {
  iso2: string;
  iso3: string;
  name: string;
  region: string;
}

export interface CoverageRule {
  readonly required: readonly MetricId[];
  readonly minCountries: number;
  readonly minPerMetric: number;
}

const NAME_OVERRIDES: Readonly<Record<string, string>> = {
  CD: 'DR Congo',
  CG: 'Republic of the Congo',
  HK: 'Hong Kong',
  MO: 'Macao',
  MM: 'Myanmar',
  PS: 'Palestine',
  FM: 'Micronesia',
  CI: "Côte d'Ivoire"
};

const regionNames = new Intl.DisplayNames(['en'], { type: 'region', fallback: 'code' });

export const isoName = (iso2: string): string | null => {
  if (!/^[A-Z]{2}$/.test(iso2)) return null;
  const name = regionNames.of(iso2);
  return name && name !== iso2 ? name : null;
};

export const displayName = (iso2: string, fallback: string): string => NAME_OVERRIDES[iso2] ?? isoName(iso2) ?? fallback;

const round = (value: number, precision: number): number => {
  const factor = 10 ** precision;
  return Math.round(value * factor) / factor;
};

export const buildDataset = (
  countries: readonly SourceCountry[],
  metrics: readonly Metric[],
  observations: ReadonlyMap<MetricId, ObservationsByIso3>,
  required: readonly MetricId[]
): Dataset => {
  const records = countries.flatMap((country): CountryRecord[] => {
    const entries = metrics.flatMap((metric): [MetricId, Observation][] => {
      const found = observations.get(metric.id)?.get(country.iso3);
      return found ? [[metric.id, { value: round(found.value, metric.precision), year: found.year }]] : [];
    });
    const values = Object.fromEntries(entries) as CountryRecord['metrics'];
    if (required.some(id => values[id] === undefined)) return [];
    return [{ iso2: country.iso2, iso3: country.iso3, name: country.name, region: country.region, metrics: values }];
  });
  records.sort((a, b) => a.name.localeCompare(b.name, 'en') || a.iso2.localeCompare(b.iso2));
  return { countries: records };
};

export const validateDataset = (dataset: Dataset, metrics: readonly Metric[], rule: CoverageRule): string[] => {
  const problems: string[] = [];
  const { countries } = dataset;
  if (countries.length < rule.minCountries) problems.push(`Only ${countries.length} countries (minimum ${rule.minCountries})`);
  const duplicates = countries.length - new Set(countries.map(c => c.iso2)).size;
  if (duplicates > 0) problems.push(`${duplicates} duplicate ISO codes`);
  for (const metric of metrics) {
    const covered = countries.filter(c => c.metrics[metric.id] !== undefined).length;
    if (covered < rule.minPerMetric) problems.push(`${metric.id}: only ${covered} countries (minimum ${rule.minPerMetric})`);
  }
  return problems;
};

export const serialize = (dataset: Dataset): string => `${JSON.stringify(dataset, null, 2)}\n`;

const USER_AGENT = 'mycountryquiz-data-updater (+https://github.com/mycountryquiz/mycountryquiz.github.io)';
const TIMEOUT_MS = 60_000;
const ATTEMPTS = 4;

const isRetryable = (status: number): boolean => status === 429 || status >= 500;
const sleep = (ms: number) => new Promise(resolve => setTimeout(resolve, ms));

export const fetchText = async (url: string): Promise<string> => {
  let lastError: unknown;
  for (let attempt = 1; attempt <= ATTEMPTS; attempt++) {
    try {
      const response = await fetch(url, {
        headers: { 'User-Agent': USER_AGENT },
        signal: AbortSignal.timeout(TIMEOUT_MS)
      });
      if (response.ok) return await response.text();
      lastError = new Error(`${response.status} ${response.statusText} for ${url}`);
      if (!isRetryable(response.status)) break;
    } catch (error) {
      lastError = error;
    }
    if (attempt < ATTEMPTS) await sleep(1000 * 2 ** attempt);
  }
  throw lastError instanceof Error ? lastError : new Error(`Failed to fetch ${url}`);
};

export const fetchJson = async <T>(url: string): Promise<T> => JSON.parse(await fetchText(url)) as T;

import type { Observation } from '../../src/types.ts';

export type ObservationsByIso3 = Map<string, Required<Observation>>;

export interface RawObservation {
  iso3: string;
  year: number;
  value: number;
}

export const latestByCountry = (rows: Iterable<RawObservation>, minYear: number): ObservationsByIso3 => {
  const latest: ObservationsByIso3 = new Map();
  for (const { iso3, year, value } of rows) {
    if (!iso3 || !Number.isFinite(value) || !Number.isInteger(year) || year < minYear) continue;
    const current = latest.get(iso3);
    if (!current || year > current.year) latest.set(iso3, { value, year });
  }
  return latest;
};

import { fetchText } from './http.ts';
import type { RawObservation } from './observations.ts';

const KEY_COLUMNS = ['Entity', 'Code', 'Year'] as const;

export const parseCsv = (text: string): string[][] => {
  const rows: string[][] = [];
  let row: string[] = [];
  let field = '';
  let quoted = false;
  for (let i = 0; i < text.length; i++) {
    const char = text[i]!;
    if (quoted) {
      if (char === '"' && text[i + 1] === '"') {
        field += '"';
        i++;
      } else if (char === '"') {
        quoted = false;
      } else {
        field += char;
      }
    } else if (char === '"') {
      quoted = true;
    } else if (char === ',') {
      row.push(field);
      field = '';
    } else if (char === '\n' || char === '\r') {
      if (char === '\r' && text[i + 1] === '\n') i++;
      row.push(field);
      rows.push(row);
      row = [];
      field = '';
    } else {
      field += char;
    }
  }
  if (field !== '' || row.length > 0) {
    row.push(field);
    rows.push(row);
  }
  return rows.filter(r => r.some(cell => cell !== ''));
};

export const toRawObservations = (rows: readonly string[][]): RawObservation[] => {
  const [header, ...data] = rows;
  if (!header) return [];
  const columnOf = (name: string): number => {
    const index = header.indexOf(name);
    if (index === -1) throw new Error(`OWID CSV is missing column "${name}": ${header.join(',')}`);
    return index;
  };
  const code = columnOf('Code');
  const year = columnOf('Year');
  const value = header.findIndex(name => !(KEY_COLUMNS as readonly string[]).includes(name));
  if (value === -1) throw new Error(`OWID CSV has no value column: ${header.join(',')}`);
  return data.flatMap(r => {
    const iso3 = r[code] ?? '';
    const raw = r[value] ?? '';
    return /^[A-Z]{3}$/.test(iso3) && raw !== '' ? [{ iso3, year: Number(r[year]), value: Number(raw) }] : [];
  });
};

export const fetchChart = async (chart: string): Promise<RawObservation[]> => {
  const params = new URLSearchParams({ v: '1', csvType: 'full', useColumnShortNames: 'false' });
  return toRawObservations(parseCsv(await fetchText(`https://ourworldindata.org/grapher/${encodeURIComponent(chart)}.csv?${params}`)));
};

import { fetchJson } from './http.ts';
import type { RawObservation } from './observations.ts';

const API = 'https://api.worldbank.org/v2';
const PAGE_SIZE = 20_000;

interface PageInfo {
  page: number;
  pages: number;
}

type ApiResponse<T> = [PageInfo, T[] | null] | [{ message: unknown }];

export interface WorldBankCountry {
  id: string;
  iso2Code: string;
  name: string;
  region: { id: string; value: string };
}

interface IndicatorRow {
  countryiso3code: string;
  date: string;
  value: number | null;
}

const fetchAllPages = async <T>(path: string, query: Record<string, string>): Promise<T[]> => {
  const rows: T[] = [];
  for (let page = 1, pages = 1; page <= pages; page++) {
    const params = new URLSearchParams({ ...query, format: 'json', per_page: String(PAGE_SIZE), page: String(page) });
    const body = await fetchJson<ApiResponse<T>>(`${API}/${path}?${params}`);
    const [info, data] = body;
    if (!('pages' in info)) throw new Error(`World Bank API error for ${path}: ${JSON.stringify(info.message)}`);
    pages = info.pages;
    rows.push(...(data ?? []));
  }
  return rows;
};

export const isAggregate = (country: WorldBankCountry): boolean => country.region.id === 'NA' || country.region.value.trim() === 'Aggregates';

export const fetchCountries = async (): Promise<WorldBankCountry[]> =>
  (await fetchAllPages<WorldBankCountry>('country', {})).filter(c => !isAggregate(c));

export const toRawObservations = (rows: readonly IndicatorRow[]): RawObservation[] =>
  rows.flatMap(row => (row.value === null ? [] : [{ iso3: row.countryiso3code, year: Number(row.date), value: row.value }]));

export const fetchIndicator = async (indicator: string, fromYear: number, toYear: number): Promise<RawObservation[]> =>
  toRawObservations(await fetchAllPages<IndicatorRow>(`country/all/indicator/${encodeURIComponent(indicator)}`, { date: `${fromYear}:${toYear}` }));

import { readFile, writeFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import { METRICS } from '../src/metrics.ts';
import type { Metric, MetricId } from '../src/types.ts';
import { buildDataset, displayName, isoName, serialize, validateDataset, type CoverageRule, type SourceCountry } from './lib/dataset.ts';
import { latestByCountry, type ObservationsByIso3, type RawObservation } from './lib/observations.ts';
import { fetchChart } from './lib/owid.ts';
import { fetchCountries, fetchIndicator } from './lib/world-bank.ts';

const OUTPUT = fileURLToPath(new URL('../src/data/countries.json', import.meta.url));
const LOOKBACK_YEARS = 20;
const COVERAGE: CoverageRule = { required: ['population', 'area'], minCountries: 180, minPerMetric: 60 };

const fetchMetric = (metric: Metric, fromYear: number, toYear: number): Promise<RawObservation[]> => {
  const { source } = metric;
  switch (source.kind) {
    case 'worldBank':
      return fetchIndicator(source.indicator, fromYear, toYear);
    case 'owid':
      return fetchChart(source.chart);
  }
};

const loadCountries = async (): Promise<SourceCountry[]> =>
  (await fetchCountries())
    .filter(c => isoName(c.iso2Code) !== null)
    .map(c => ({ iso2: c.iso2Code, iso3: c.id, name: displayName(c.iso2Code, c.name.trim()), region: c.region.value.trim() }));

const loadObservations = async (fromYear: number, toYear: number): Promise<Map<MetricId, ObservationsByIso3>> => {
  const observations = new Map<MetricId, ObservationsByIso3>();
  for (const metric of METRICS) {
    const latest = latestByCountry(await fetchMetric(metric, fromYear, toYear), fromYear);
    console.log(`${metric.id.padEnd(20)} ${String(latest.size).padStart(4)} countries`);
    observations.set(metric.id, latest);
  }
  return observations;
};

const readExisting = async (): Promise<string | null> => {
  try {
    return await readFile(OUTPUT, 'utf8');
  } catch {
    return null;
  }
};

const main = async (): Promise<void> => {
  const toYear = new Date().getUTCFullYear();
  const fromYear = toYear - LOOKBACK_YEARS;
  const [countries, observations] = await Promise.all([loadCountries(), loadObservations(fromYear, toYear)]);
  const dataset = buildDataset(countries, METRICS, observations, COVERAGE.required);

  const problems = validateDataset(dataset, METRICS, COVERAGE);
  if (problems.length > 0) throw new Error(`Dataset failed validation:\n- ${problems.join('\n- ')}`);

  const output = serialize(dataset);
  if (output === (await readExisting())) {
    console.log('Data unchanged.');
    return;
  }
  await writeFile(OUTPUT, output, 'utf8');
  console.log(`Wrote ${dataset.countries.length} countries to ${OUTPUT}`);
};

main().catch(error => {
  console.error(error instanceof Error ? error.message : error);
  process.exitCode = 1;
});

export type Cue = 'tap' | 'correct' | 'wrong';

interface Tone {
  readonly frequency: number;
  readonly start: number;
  readonly duration: number;
  readonly wave: OscillatorType;
}

const TONES: Record<Cue, readonly Tone[]> = {
  tap: [{ frequency: 660, start: 0, duration: 0.04, wave: 'triangle' }],
  correct: [
    { frequency: 660, start: 0, duration: 0.09, wave: 'sine' },
    { frequency: 990, start: 0.09, duration: 0.16, wave: 'sine' }
  ],
  wrong: [
    { frequency: 220, start: 0, duration: 0.12, wave: 'square' },
    { frequency: 165, start: 0.12, duration: 0.2, wave: 'square' }
  ]
};

const VIBRATIONS: Record<Cue, VibratePattern> = {
  tap: 12,
  correct: [25, 50, 25],
  wrong: [80, 60, 80]
};

const PEAK_GAIN = 0.08;

export class Feedback {
  #context: AudioContext | null = null;
  enabled: boolean;

  constructor(enabled: boolean) {
    this.enabled = enabled;
  }

  play(cue: Cue): void {
    if (!this.enabled) return;
    this.#vibrate(cue);
    this.#sound(cue);
  }

  #vibrate(cue: Cue): void {
    if ('vibrate' in navigator) navigator.vibrate(VIBRATIONS[cue]);
  }

  #sound(cue: Cue): void {
    const context = this.#audio();
    if (!context) return;
    const now = context.currentTime;
    for (const tone of TONES[cue]) {
      const oscillator = context.createOscillator();
      const gain = context.createGain();
      const begin = now + tone.start;
      const end = begin + tone.duration;
      oscillator.type = tone.wave;
      oscillator.frequency.setValueAtTime(tone.frequency, begin);
      gain.gain.setValueAtTime(0.0001, begin);
      gain.gain.exponentialRampToValueAtTime(PEAK_GAIN, begin + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.0001, end);
      oscillator.connect(gain).connect(context.destination);
      oscillator.start(begin);
      oscillator.stop(end + 0.02);
    }
  }

  #audio(): AudioContext | null {
    if (typeof AudioContext === 'undefined') return null;
    this.#context ??= new AudioContext();
    if (this.#context.state === 'suspended') void this.#context.resume();
    return this.#context;
  }
}

import type { Country, Metric } from './types.ts';

export interface Round {
  readonly metric: Metric;
  readonly first: Country;
  readonly second: Country;
  readonly answer: Country;
}

export type Random = () => number;

const MAX_ATTEMPTS = 200;

export const valueOf = (metric: Metric, country: Country): number | undefined => country.metrics[metric.id]?.value;

export const isDistinguishable = (metric: Metric, a: Country, b: Country): boolean => {
  const va = valueOf(metric, a);
  const vb = valueOf(metric, b);
  return va !== undefined && vb !== undefined && va !== vb && metric.format(va) !== metric.format(vb);
};

const candidatesFor = (metric: Metric, countries: readonly Country[]): Country[] =>
  countries.filter(c => valueOf(metric, c) !== undefined);

export const isPlayable = (metric: Metric, countries: readonly Country[]): boolean =>
  new Set(candidatesFor(metric, countries).map(c => metric.format(valueOf(metric, c)!))).size >= 2;

export const createRound = (metric: Metric, countries: readonly Country[], random: Random = Math.random): Round | null => {
  const pool = candidatesFor(metric, countries);
  if (pool.length < 2) return null;

  const pick = () => pool[Math.floor(random() * pool.length)]!;
  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
    const first = pick();
    const second = pick();
    if (first.iso2 !== second.iso2 && isDistinguishable(metric, first, second)) {
      const answer = valueOf(metric, first)! > valueOf(metric, second)! ? first : second;
      return { metric, first, second, answer };
    }
  }
  return null;
};

export interface Settings {
  selectedQuestionId: string;
  feedbackEnabled: boolean;
}

export type PersistedState = Stats & Settings;

const STORAGE_KEY = 'countryQuizState';

export const EMPTY_STATS: Readonly<Stats> = { correctAnswers: 0, totalQuestions: 0, currentStreak: 0, bestStreak: 0 };

const DEFAULTS: Readonly<PersistedState> = { ...EMPTY_STATS, selectedQuestionId: DEFAULT_METRIC_ID, feedbackEnabled: true };

const count = (value: unknown, fallback: number): number =>
  typeof value === 'number' && Number.isInteger(value) && value >= 0 ? value : fallback;

export const loadState = (): PersistedState => {
  try {
    const parsed: Record<string, unknown> = JSON.parse(localStorage.getItem(STORAGE_KEY) ?? '{}');
    return {
      correctAnswers: count(parsed['correctAnswers'], DEFAULTS.correctAnswers),
      totalQuestions: count(parsed['totalQuestions'], DEFAULTS.totalQuestions),
      currentStreak: count(parsed['currentStreak'], DEFAULTS.currentStreak),
      bestStreak: count(parsed['bestStreak'], DEFAULTS.bestStreak),
      selectedQuestionId: typeof parsed['selectedQuestionId'] === 'string' ? parsed['selectedQuestionId'] : DEFAULTS.selectedQuestionId,
      feedbackEnabled: typeof parsed['feedbackEnabled'] === 'boolean' ? parsed['feedbackEnabled'] : DEFAULTS.feedbackEnabled
    };
  } catch {
    return { ...DEFAULTS };
  }
};

export const saveState = (state: PersistedState): void => {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  } catch (e) {
    console.warn('Failed to save state:', e);
  }
};

const metric = findMetric('literacy')!;

const country = (iso2: string, literacy?: number): Country => ({
  iso2,
  name: iso2,
  region: 'Test',
  flag: '',
  metrics: literacy === undefined ? {} : { literacy: { value: literacy } }
});

  let i = 0;
  return () => values[i++ % values.length]!;
};

describe('isDistinguishable', () => {
  it('rejects equal values', () => assert.equal(isDistinguishable(metric, country('AA', 99), country('BB', 99)), false));
  it('rejects values that format identically', () => assert.equal(isDistinguishable(metric, country('AA', 99.01), country('BB', 99.04)), false));
  it('rejects missing values', () => assert.equal(isDistinguishable(metric, country('AA'), country('BB', 50)), false));
  it('accepts distinct values', () => assert.equal(isDistinguishable(metric, country('AA', 98), country('BB', 99)), true));
});

describe('createRound', () => {
  it('picks the higher value as the answer', () => {
    const round = createRound(metric, [country('AA', 40), country('BB', 90)], sequence(0, 0.9));
    assert.ok(round);
    assert.equal(round.answer.iso2, 'BB');
  });

  it('never pairs a country with itself or a tie', () => {
    const pool = [country('AA', 50), country('BB', 50), country('CC', 70)];
    for (let i = 0; i < 200; i++) {
      const round = createRound(metric, pool);
      assert.ok(round);
      assert.notEqual(round.first.iso2, round.second.iso2);
      assert.ok(isDistinguishable(metric, round.first, round.second));
    }
  });

  it('returns null when no valid pair exists', () => {
    assert.equal(createRound(metric, [country('AA', 50), country('BB', 50)]), null);
    assert.equal(createRound(metric, [country('AA', 50), country('BB')]), null);
  });
});

describe('isPlayable', () => {

  it('requires at least two distinguishable values', () => {
    assert.equal(isPlayable(m, [country('AA', 50), country('BB', 50)]), false);
    assert.equal(isPlayable(m, [country('AA', 50), country('BB', 60)]), true);

# 🌍 Country Quiz

A client-side country comparison quiz game. Test your knowledge of world geography, economics, and demographics by comparing countries across various metrics.

**Play now: [mycountryquiz.github.io](https://mycountryquiz.github.io)**

## Features

- **13 Quiz Categories**
  - Population
  - Area (km²)
  - GDP (nominal, current US$)
  - GDP per Capita
  - GNI per Capita (PPP)
  - Population Density
  - Adult Literacy Rate
  - Human Development Index (HDI)
  - Life Expectancy
  - Fertility Rate
  - Urbanization (share of urban population)
  - Internet Use (share of population)
  - Forest Cover (share of land area)

- **Every country with World Bank coverage** (~200 economies), each value shown with the year it was measured
- **Automatically refreshed data** via a weekly GitHub Actions workflow
- **No ties** — pairs with equal (or identically displayed) values are never asked
- **Score Tracking**
  - Current score (correct/total), current streak, best streak
  - All stats persist in localStorage
- **Audible and tactile feedback**
  - Short synthesized tones (Web Audio API) and vibration (Vibration API) on taps, correct and wrong answers
  - Toggle with the 🔊/🔇 button; the choice is remembered
  - Vibration is unavailable on iOS Safari; sound still works
- **Progressive Web App (PWA)**
  - Installable ("Add to Home Screen")
  - Works offline after first load (network-first service worker, same-origin only)
- **Accessible** — real buttons, keyboard navigation, visible focus, reduced-motion support
- **Fully Client-Side**
  - No server, no data collection, no third-party requests at runtime
  - Statistics are bundled into the app at build time

## Tech Stack

- **TypeScript 7** (native compiler)
- **Vite 8** (Rolldown-based bundler)
- **Vanilla TypeScript** - No framework dependencies
- **Node.js test runner** - Unit tests with no extra dependencies
- **GitHub Actions + GitHub Pages** - CI, scheduled data refresh, deployment
- **Dependabot** - Weekly npm and GitHub Actions updates

## Data

### Sources

| Category | Source | Indicator |
|----------|--------|-----------|
| Population | World Bank WDI | `SP.POP.TOTL` |
| Area | World Bank WDI | `AG.SRF.TOTL.K2` |
| GDP | World Bank WDI | `NY.GDP.MKTP.CD` |
| GDP per Capita | World Bank WDI | `NY.GDP.PCAP.CD` |
| GNI per Capita (PPP) | World Bank WDI | `NY.GNP.PCAP.PP.CD` |
| Population Density | World Bank WDI | `EN.POP.DNST` |
| Adult Literacy Rate | World Bank WDI (UNESCO UIS) | `SE.ADT.LITR.ZS` |
| HDI | UNDP Human Development Report via Our World in Data | `human-development-index` |
| Life Expectancy | World Bank WDI | `SP.DYN.LE00.IN` |
| Fertility Rate | World Bank WDI | `SP.DYN.TFRT.IN` |
| Urbanization | World Bank WDI | `SP.URB.TOTL.IN.ZS` |
| Internet Use | World Bank WDI (ITU) | `IT.NET.USER.ZS` |
| Forest Cover | World Bank WDI (FAO) | `AG.LND.FRST.ZS` |

- [World Bank Open Data](https://data.worldbank.org/) (CC BY 4.0)
- [UNDP Human Development Reports](https://hdr.undp.org/) via [Our World in Data](https://ourworldindata.org/grapher/human-development-index) (CC BY 4.0)

For every country and category the most recent non-empty value from the last 20 years is used, and the year is shown next to the value. Country names come from the Unicode CLDR (via `Intl.DisplayNames`) with a few short-form overrides; flags are derived from ISO 3166-1 codes. Regional aggregates and entries without population and area are excluded.

### Automatic updates

`.github/workflows/update-data.yml` runs every Monday (and on demand from the Actions tab):

1. Runs `yarn update-data`, which fetches every metric, keeps the latest value per country, rounds to each metric's precision and writes a deterministically sorted `src/data/countries.json`.
2. Rejects the result if coverage looks wrong (fewer than 180 countries, any category under 60 countries, duplicate codes) so a broken upstream response never reaches the site.
3. Runs tests and a full build.
4. Commits only when the data actually changed, then calls `deploy.yml` to publish that exact commit.

The workflow pushes to `main` with `GITHUB_TOKEN`; if branch protection blocks this, allow GitHub Actions to bypass it or switch the commit step to open a pull request.

To add a category, add an entry to `METRICS` in `src/metrics.ts` with a World Bank indicator or OWID chart slug, then run `yarn update-data`.

## Privacy

- No analytics, cookies, fingerprinting, or external requests at runtime
- localStorage holds only score, streaks, selected category and the feedback toggle
- The Web Audio API is used only to play tones; nothing is read back
- The service worker caches same-origin files on your device and never contacts other hosts

#!/bin/sh
# Dumps all project code to a single file for LLM context
# Respects .gitignore patterns
# Output: docs/llm/dump.txt

set -e

OUTPUT_DIR="docs/llm"
OUTPUT_FILE="$OUTPUT_DIR/dump.txt"

# Create output directory
mkdir -p "$OUTPUT_DIR"

# Clear/create output file
> "$OUTPUT_FILE"

echo "Dumping project code to $OUTPUT_FILE..."

# Use git ls-files to get tracked files (respects .gitignore)
# Falls back to find if not a git repo
if git rev-parse --git-dir > /dev/null 2>&1; then
    FILES=$(git ls-files --cached --others --exclude-standard 2>/dev/null)
else
    echo "Warning: Not a git repository, using find (may include ignored files)"
    FILES=$(find . -type f \
        ! -path './node_modules/*' \
        ! -path './dist/*' \
        ! -path './.git/*' \
        ! -path './docs/llm/*' \
        ! -name '*.lock' \
        ! -name 'package-lock.json' \
        | sed 's|^\./||')
fi

# File extensions to include
is_code_file() {
    case "$1" in
        *.ts|*.tsx|*.js|*.jsx|*.mjs|*.cjs) return 0 ;;
        *.css|*.scss|*.less) return 0 ;;
        *.html|*.htm) return 0 ;;
        *.json) return 0 ;;
        *.md|*.txt) return 0 ;;
        *.sh) return 0 ;;
        *.yaml|*.yml) return 0 ;;
        *.toml) return 0 ;;
        *.svg) return 0 ;;
        *) return 1 ;;
    esac
}

# Skip certain files
should_skip() {
    case "$1" in
        package-lock.json|pnpm-lock.yaml|yarn.lock) return 0 ;;
        *.min.js|*.min.css) return 0 ;;
        docs/llm/*) return 0 ;;
        *) return 1 ;;
    esac
}

count=0

echo "$FILES" | while IFS= read -r file; do
    # Skip empty lines
    [ -z "$file" ] && continue
    
    # Skip if not a regular file
    [ ! -f "$file" ] && continue
    
    # Skip non-code files
    is_code_file "$file" || continue
    
    # Skip certain files
    should_skip "$file" && continue
    
    # Get file size
    size=$(wc -c < "$file" | tr -d ' ')
    
    # Skip files larger than 100KB
    [ "$size" -gt 102400 ] && continue
    
    echo "================================================================================" >> "$OUTPUT_FILE"
    echo "FILE: $file" >> "$OUTPUT_FILE"
    echo "================================================================================" >> "$OUTPUT_FILE"
    cat "$file" >> "$OUTPUT_FILE"
    echo "" >> "$OUTPUT_FILE"
    echo "" >> "$OUTPUT_FILE"
    
    count=$((count + 1))
done

# Count files (recount since subshell)
file_count=$(grep -c "^FILE:" "$OUTPUT_FILE" 2>/dev/null || echo "0")
total_size=$(wc -c < "$OUTPUT_FILE" | tr -d ' ')

echo "Done! Dumped $file_count files ($total_size bytes) to $OUTPUT_FILE"

import raw from './data/countries.json' with { type: 'json' };
import type { Country, Dataset } from './types.ts';

const REGIONAL_INDICATOR_OFFSET = 0x1f1e6 - 'A'.charCodeAt(0);

export const toFlag = (iso2: string): string =>
  String.fromCodePoint(...[...iso2.toUpperCase()].map(c => c.charCodeAt(0) + REGIONAL_INDICATOR_OFFSET));

const dataset = raw as Dataset;

export const COUNTRIES: readonly Country[] = dataset.countries.map(c => ({ ...c, flag: toFlag(c.iso2) }));

import './style.css';
import { COUNTRIES } from './data.ts';
import { Feedback } from './feedback.ts';
import { DEFAULT_METRIC_ID, METRICS, findMetric } from './metrics.ts';
import { createRound, isPlayable, valueOf, type Round } from './quiz.ts';
import { EMPTY_STATS, loadState, saveState, type PersistedState } from './storage.ts';
import type { Country, Metric } from './types.ts';

const TEMPLATE = `
  <main class="game-container">
    <h1 class="game-title">🌍 Country Quiz</h1>
    <div class="controls">
      <select id="question-select" aria-label="Question category"></select>
      <button id="feedback-btn" type="button"></button>
      <button id="reset-btn" type="button">Reset</button>
    </div>
    <div class="score-display">
      <span class="score">Score: <span id="score">0/0</span></span>
      <span id="streak"></span>
      <span id="best-streak"></span>
    </div>
    <p id="question-text" class="question-text" aria-live="polite"></p>
    <div class="comparison-container">
      <button id="country1" type="button" class="country-card"></button>
      <div class="vs-badge" aria-hidden="true">VS</div>
      <button id="country2" type="button" class="country-card"></button>
    </div>
    <div id="result-message" class="result-message hidden" role="status"></div>
    <button id="next-btn" type="button" class="next-btn hidden">Next →</button>
    <footer class="attribution">Data: World Bank WDI · UNDP HDR via Our World in Data (CC BY 4.0)</footer>
  </main>
`;

type CardState = 'pending' | 'revealed' | 'correct' | 'wrong';

const byId = <T extends HTMLElement>(id: string): T => {
  const element = document.getElementById(id);
  if (!element) throw new Error(`Missing element #${id}`);
  return element as T;
};

class QuizApp {
  readonly #state: PersistedState;
  readonly #feedback: Feedback;
  readonly #metrics: readonly Metric[];
  #round: Round | null = null;
  #answered = false;

  readonly #question: HTMLElement;
  readonly #cards: readonly [HTMLButtonElement, HTMLButtonElement];
  readonly #result: HTMLElement;
  readonly #next: HTMLButtonElement;
  readonly #score: HTMLElement;
  readonly #streak: HTMLElement;
  readonly #best: HTMLElement;
  readonly #select: HTMLSelectElement;
  readonly #feedbackToggle: HTMLButtonElement;

  constructor(root: HTMLElement) {
    root.innerHTML = TEMPLATE;
    this.#state = loadState();
    this.#feedback = new Feedback(this.#state.feedbackEnabled);
    this.#metrics = METRICS.filter(m => isPlayable(m, COUNTRIES));

    this.#question = byId('question-text');
    this.#cards = [byId('country1'), byId('country2')];
    this.#result = byId('result-message');
    this.#next = byId('next-btn');
    this.#score = byId('score');
    this.#streak = byId('streak');
    this.#best = byId('best-streak');
    this.#select = byId('question-select');
    this.#feedbackToggle = byId('feedback-btn');

    this.#bind(byId('reset-btn'));
    this.#renderSelect();
    this.#renderFeedbackToggle();
    this.#renderScore();
    this.#nextRound();
  }

  #bind(reset: HTMLButtonElement): void {
    this.#cards.forEach((card, index) => card.addEventListener('click', () => this.#answer(index)));
    this.#next.addEventListener('click', () => {
      this.#feedback.play('tap');
      this.#nextRound();
    });
    reset.addEventListener('click', () => {
      this.#feedback.play('tap');
      this.#reset();
    });
    this.#feedbackToggle.addEventListener('click', () => this.#toggleFeedback());
    this.#select.addEventListener('change', () => {
      this.#feedback.play('tap');
      this.#state.selectedQuestionId = this.#select.value;
      this.#persist();
      this.#nextRound();
    });
  }

  get #metric(): Metric {
    return this.#metrics.find(m => m.id === this.#state.selectedQuestionId)
      ?? findMetric(DEFAULT_METRIC_ID)
      ?? METRICS[0]!;
  }

  #nextRound(): void {
    this.#round = createRound(this.#metric, COUNTRIES);
    this.#answered = false;
    this.#result.className = 'result-message hidden';
    this.#next.classList.add('hidden');

    if (!this.#round) {
      this.#question.textContent = 'Not enough data for this question type.';
      this.#cards.forEach(card => (card.hidden = true));
      return;
    }

    this.#question.textContent = this.#round.metric.question;
    this.#renderCard(this.#cards[0], this.#round.first, 'pending');
    this.#renderCard(this.#cards[1], this.#round.second, 'pending');
  }

  #answer(index: number): void {
    const round = this.#round;
    if (this.#answered || !round) return;
    this.#answered = true;

    const selected = index === 0 ? round.first : round.second;
    const correct = selected.iso2 === round.answer.iso2;
    this.#feedback.play(correct ? 'correct' : 'wrong');
    this.#record(correct);

    this.#result.textContent = correct ? '✓ Correct!' : `✗ Wrong! ${round.answer.name} was correct.`;
    this.#result.className = `result-message ${correct ? 'correct' : 'wrong'}`;

    const outcome: CardState = correct ? 'correct' : 'wrong';
    this.#renderCard(this.#cards[0], round.first, index === 0 ? outcome : 'revealed');
    this.#renderCard(this.#cards[1], round.second, index === 1 ? outcome : 'revealed');

    this.#next.classList.remove('hidden');
    this.#next.focus({ preventScroll: true });
    this.#renderScore();
    this.#persist();
  }

  #record(correct: boolean): void {
    this.#state.totalQuestions++;
    if (correct) {
      this.#state.correctAnswers++;
      this.#state.currentStreak++;
      this.#state.bestStreak = Math.max(this.#state.bestStreak, this.#state.currentStreak);
    } else {
      this.#state.currentStreak = 0;
    }
  }

  #reset(): void {
    if (!confirm('Reset all scores?')) return;
    Object.assign(this.#state, EMPTY_STATS);
    this.#persist();
    this.#renderScore();
    this.#nextRound();
  }

  #toggleFeedback(): void {
    this.#state.feedbackEnabled = !this.#state.feedbackEnabled;
    this.#feedback.enabled = this.#state.feedbackEnabled;
    this.#feedback.play('tap');
    this.#renderFeedbackToggle();
    this.#persist();
  }

  #persist(): void {
    saveState(this.#state);
  }

  #renderCard(card: HTMLButtonElement, country: Country, state: CardState): void {
    const revealed = state !== 'pending';
    card.hidden = false;
    card.disabled = revealed;
    card.className = state === 'correct' || state === 'wrong' ? `country-card ${state}` : 'country-card';

    const flag = document.createElement('span');
    flag.className = 'country-flag';
    flag.textContent = country.flag;
    flag.setAttribute('aria-hidden', 'true');

    const name = document.createElement('span');
    name.className = 'country-name';
    name.textContent = country.name;

    card.replaceChildren(flag, name);
    if (revealed && this.#round) card.append(this.#valueElement(this.#round.metric, country));
  }

  #valueElement(metric: Metric, country: Country): HTMLElement {
    const value = valueOf(metric, country);
    const year = country.metrics[metric.id]?.year;
    const element = document.createElement('span');
    element.className = 'value-reveal';
    element.textContent = value === undefined ? 'N/A' : `${metric.format(value)}${year ? ` (${year})` : ''}`;
    return element;
  }

  #renderSelect(): void {
    this.#select.replaceChildren(...this.#metrics.map(m => new Option(m.label, m.id, false, m.id === this.#metric.id)));
  }

  #renderFeedbackToggle(): void {
    const on = this.#state.feedbackEnabled;
    this.#feedbackToggle.textContent = on ? '🔊' : '🔇';
    this.#feedbackToggle.setAttribute('aria-pressed', String(on));
    this.#feedbackToggle.setAttribute('aria-label', on ? 'Sound and vibration on' : 'Sound and vibration off');
    this.#feedbackToggle.title = this.#feedbackToggle.getAttribute('aria-label')!;
  }

  #renderScore(): void {
    const s = this.#state;
    this.#score.textContent = `${s.correctAnswers}/${s.totalQuestions}`;
    this.#streak.textContent = s.currentStreak > 0 ? `🔥 ${s.currentStreak}` : '';
    this.#best.textContent = s.bestStreak > 0 ? `⭐ Best: ${s.bestStreak}` : '';
  }
}

const registerServiceWorker = (): void => {
  if (import.meta.env.PROD && 'serviceWorker' in navigator) {
    void navigator.serviceWorker.register(`${import.meta.env.BASE_URL}sw.js`).catch(e => console.warn('Service worker registration failed:', e));
  }
};

document.addEventListener('DOMContentLoaded', () => {
  new QuizApp(byId('app'));
  registerServiceWorker();
});
